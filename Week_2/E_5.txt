class Solution {
    public boolean isUgly(int n) {
        boolean flag = true;
        boolean isPrime = true;
        if(n == 1) return true;
        if(n <= 0) return false;

        int[] factors = {2, 3, 5};
        for(int i = 0; i < factors.length; i++){
            while(n % factors[i] == 0){
                n = n / factors[i];
            }
        }
        if(n != 1) flag = false;
        return flag;
    }
}
